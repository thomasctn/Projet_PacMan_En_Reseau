\begin{frame}[fragile]{Format des Messages}
\begin{columns}
\column{0.60\textwidth}
\textbf{Messages : Structures C++}
\begin{itemize}
    \item Connues par le client et le serveur
    \item Contiennent des données souvent regroupées dans des structures \textit{communes}
    \item Un champ \textcolor{blue}{\texttt{gf::id}} pour les identifier lors de la désérialisation
    \item Nomenclature spécifique :
    \begin{itemize}
            \item \textit{ServerXXX} = envoyée par le serveur
            \item \textit{ClientXXX} = envoyée par le client
    \end{itemize}
\end{itemize}
\vspace{0.25cm}
\textbf{Sérialisation/désérialisation}
\begin{itemize}
    \item Nécessaire pour l'envoi/réception
    \item \textbf{Opétateur |} avec un type \textit{Archive} pour chaque structure
\end{itemize}

\column{0.35\textwidth}
\begin{lstlisting}[linewidth=4cm]
struct PlayerData
{
    uint32_t id;
    float x, y;
    std::string name;
    PlayerRole role;
    int score;
    bool ready = false;
    unsigned int hp = 0;
};
\end{lstlisting}
\begin{lstlisting}[linewidth=4cm]
struct ServerListRoomPlayers
{
  static constexpr gf::Id type = "ServerListRoomPlayers"_id;
  std::vector<PlayerData> players;
};
\end{lstlisting}
\begin{lstlisting}[linewidth=4cm]
template <typename Archive>
Archive &operator|(Archive &ar, ServerListRoomPlayers &data)
{
  return ar | data.players;
}
\end{lstlisting}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Structure des échanges}
    \begin{columns}

    \column{0.55\textwidth}
    \textbf{Connexion}
    \begin{itemize}
        \item \textbf{Client} : Demande de connexion
        \item \textbf{Serveur} : Assignation d'un id et renvoi de l'id + liste des salons
    \end{itemize}
    \textbf{Lobby}
    \begin{itemize}
        \item \textbf{Client} : Demande de création/connexion à un salon
        \item \textbf{Serveur} : Traitement et envoi de la réponse et \textbf{broadcast} de la liste des joueurs/paramètres
    \end{itemize}
    \column{0.4\textwidth}
    \includegraphics[width=\textwidth]{UML/PacMan_UML_Reseau_connexion.png}
    \end{columns}

\end{frame}

\begin{frame}[fragile]{Structure des échanges}
    \begin{columns}
    \column{0.55\textwidth}
    \textbf{Salon}
    \begin{itemize}
        \item \textbf{Client} : Demande de changement de rôle/status/paramètres
        \item \textbf{Serveur} : Traitement et \textbf{broadcast} de la liste des joueurs/paramètres
    \end{itemize}
        Si tous les joueurs son prêt : \textbf{Broadcast} début de partie ou erreur\\
    \textbf{En jeu}
    \begin{itemize}
        \item \textbf{Client} : Demande de déplacement
        \item \textbf{Serveur} : Vérifie la validité, gère les conséquences puis \textbf{broadcast} de l'état du jeu
    \end{itemize}
     Si fin de jeu : \textbf{Broadcast} de la fin du jeu et de la raison
